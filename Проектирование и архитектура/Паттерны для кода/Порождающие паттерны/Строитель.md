
Этот паттерн берёт на себя логику по созданию объекта, валидацию объекта с помощью реализации итогого метода Build а также улучшается читабельность кода с помощью создания цепочки вызова методов.

`public class EmailBuilder {` 
	`private readonly EmailMessage _email = new();`
	`public EmailBuilder From(string from)` 
	`{`
		 `_email.From = from;` 
		 `return this;`
	`}`
`}`

`var email = new EmailBuilder()`
    `.Attach("file.pdf")`
    `.Body("text")`
    `.Build();`
